----- Fundamentals: run a single Pod imperatively, expose it, then delete it -----
$ kubectl run hello --image=nginx:alpine --port=80
pod/hello created
$ kubectl get pod hello -o wide
NAME    READY   STATUS    RESTARTS   AGE   IP          NODE     NOMINATED NODE   READINESS GATES
hello   1/1     Running   0          5s    10.42.0.9   colima   <none>           <none>
$ kubectl expose pod hello --port=80 --name=hello-svc
service/hello-svc exposed
$ kubectl get svc hello-svc
NAME        TYPE        CLUSTER-IP     EXTERNAL-IP   PORT(S)   AGE
hello-svc   ClusterIP   10.43.239.41   <none>        80/TCP    1s
